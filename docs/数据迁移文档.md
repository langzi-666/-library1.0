# 图书管理系统数据迁移文档

## 1. 迁移概述

### 1.1 迁移目的
将现有图书管理系统的数据从旧系统迁移到新系统，确保数据的完整性和一致性。

### 1.2 迁移范围
- 用户数据
- 图书数据
- 图书分类数据
- 借阅记录数据
- 操作日志数据
- 系统设置数据

### 1.3 迁移环境
- **源系统**：旧图书管理系统
- **目标系统**：新图书管理系统（library1.4）
- **数据库**：MySQL 8.0+

## 3. 数据迁移脚本

### 3.1 用户数据迁移

#### 3.1.1 迁移脚本（SQL）

```sql
-- 用户数据迁移脚本
-- 假设源表为 old_user，目标表为 user

-- 1. 迁移用户基本信息
INSERT INTO library1.4.user (
    user_id,
    username,
    password,
    email,
    phone,
    real_name,
    role,
    status,
    create_time,
    update_time
)
SELECT 
    CONCAT('U', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(ROW_NUMBER() OVER (ORDER BY old_user.id), 4, '0')) AS user_id,
    username,
    password,  -- 如果密码加密方式不同，需要转换
    email,
    phone,
    real_name,
    CASE role
        WHEN 'admin' THEN 'SYSTEM_ADMIN'
        WHEN 'librarian' THEN 'LIBRARIAN'
        WHEN 'user' THEN 'USER'
        ELSE 'USER'
    END AS role,
    CASE status
        WHEN 1 THEN 'ACTIVE'
        WHEN 0 THEN 'INACTIVE'
        ELSE 'INACTIVE'
    END AS status,
    create_time,
    update_time
FROM old_library.old_user
WHERE username IS NOT NULL;

-- 2. 验证迁移结果
SELECT COUNT(*) AS total_users FROM library1.4.user;
SELECT role, COUNT(*) AS count FROM library1.4.user GROUP BY role;
```

#### 3.1.2 迁移脚本（Python示例）

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
用户数据迁移脚本
"""
import pymysql
from datetime import datetime

# 数据库连接配置
SOURCE_DB = {
    'host': 'localhost',
    'user': 'root',
    'password': 'old_password',
    'database': 'old_library',
    'charset': 'utf8mb4'
}

TARGET_DB = {
    'host': 'localhost',
    'user': 'root',
    'password': 'new_password',
    'database': 'library1.4',
    'charset': 'utf8mb4'
}

def migrate_users():
    """迁移用户数据"""
    source_conn = pymysql.connect(**SOURCE_DB)
    target_conn = pymysql.connect(**TARGET_DB)
    
    try:
        source_cursor = source_conn.cursor(pymysql.cursors.DictCursor)
        target_cursor = target_conn.cursor()
        
        # 查询源数据
        source_cursor.execute("SELECT * FROM old_user")
        users = source_cursor.fetchall()
        
        # 生成用户ID序列
        date_prefix = datetime.now().strftime('%Y%m%d')
        user_counter = 1
        
        # 插入目标数据库
        for user in users:
            user_id = f"U{date_prefix}{user_counter:04d}"
            
            # 角色映射
            role_map = {
                'admin': 'SYSTEM_ADMIN',
                'librarian': 'LIBRARIAN',
                'user': 'USER'
            }
            role = role_map.get(user['role'], 'USER')
            
            # 状态映射
            status = 'ACTIVE' if user['status'] == 1 else 'INACTIVE'
            
            # 插入数据
            sql = """
                INSERT INTO user (
                    user_id, username, password, email, phone,
                    real_name, role, status, create_time, update_time
                ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
            """
            target_cursor.execute(sql, (
                user_id, user['username'], user['password'],
                user['email'], user['phone'], user['real_name'],
                role, status, user['create_time'], user['update_time']
            ))
            
            user_counter += 1
        
        target_conn.commit()
        print(f"成功迁移 {len(users)} 条用户数据")
        
    except Exception as e:
        target_conn.rollback()
        print(f"迁移失败: {e}")
    finally:
        source_conn.close()
        target_conn.close()

if __name__ == '__main__':
    migrate_users()
```

### 3.2 图书分类数据迁移

```sql
-- 图书分类数据迁移
-- 假设源表为 old_category，目标表为 category

-- 1. 迁移一级分类
INSERT INTO library1.4.category (
    category_id,
    category_name,
    parent_id,
    level,
    sort_order,
    description,
    create_time,
    update_time
)
SELECT 
    CONCAT('CAT', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(ROW_NUMBER() OVER (ORDER BY old_category.id), 4, '0')) AS category_id,
    category_name,
    NULL AS parent_id,
    1 AS level,
    sort_order,
    description,
    create_time,
    update_time
FROM old_library.old_category
WHERE parent_id IS NULL OR parent_id = 0
ORDER BY sort_order;

-- 2. 迁移二级分类（需要先获取一级分类的映射）
-- 创建临时映射表
CREATE TEMPORARY TABLE category_mapping (
    old_id INT,
    new_id VARCHAR(20)
);

-- 填充映射表（需要根据实际情况调整）
INSERT INTO category_mapping (old_id, new_id)
SELECT old_id, new_id FROM (
    -- 这里需要手动或通过脚本建立映射关系
    SELECT 1 AS old_id, 'CAT202412010001' AS new_id
    UNION ALL SELECT 2, 'CAT202412010002'
    -- ... 更多映射
) AS mapping;

-- 迁移二级分类
INSERT INTO library1.4.category (
    category_id,
    category_name,
    parent_id,
    level,
    sort_order,
    description,
    create_time,
    update_time
)
SELECT 
    CONCAT('CAT', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(ROW_NUMBER() OVER (ORDER BY old_category.id), 4, '0')) AS category_id,
    old_category.category_name,
    cm.new_id AS parent_id,
    2 AS level,
    old_category.sort_order,
    old_category.description,
    old_category.create_time,
    old_category.update_time
FROM old_library.old_category
JOIN category_mapping cm ON old_category.parent_id = cm.old_id
WHERE old_category.parent_id IS NOT NULL AND old_category.parent_id != 0;

-- 3. 验证迁移结果
SELECT level, COUNT(*) AS count FROM library1.4.category GROUP BY level;
```

### 3.3 图书数据迁移

```sql
-- 图书数据迁移
-- 假设源表为 old_book，目标表为 book

-- 1. 迁移图书数据
INSERT INTO library1.4.book (
    book_id,
    isbn,
    title,
    author,
    publisher,
    publish_date,
    category_id,
    price,
    stock,
    available_stock,
    status,
    description,
    cover_image,
    create_time,
    update_time
)
SELECT 
    CONCAT('BK', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(ROW_NUMBER() OVER (ORDER BY old_book.id), 4, '0')) AS book_id,
    isbn,
    title,
    author,
    publisher,
    publish_date,
    -- 需要分类ID映射（通过临时表）
    (SELECT new_id FROM category_mapping WHERE old_id = old_book.category_id) AS category_id,
    price,
    stock,
    available_stock,
    CASE status
        WHEN 'available' THEN 'AVAILABLE'
        WHEN 'borrowed' THEN 'BORROWED'
        WHEN 'damaged' THEN 'DAMAGED'
        WHEN 'offline' THEN 'OFFLINE'
        ELSE 'AVAILABLE'
    END AS status,
    description,
    cover_image,
    create_time,
    update_time
FROM old_library.old_book
WHERE isbn IS NOT NULL;

-- 2. 验证迁移结果
SELECT COUNT(*) AS total_books FROM library1.4.book;
SELECT status, COUNT(*) AS count FROM library1.4.book GROUP BY status;
```

### 3.4 借阅记录数据迁移

```sql
-- 借阅记录数据迁移
-- 假设源表为 old_borrow，目标表为 borrow_record

-- 1. 迁移借阅记录
INSERT INTO library1.4.borrow_record (
    record_id,
    user_id,
    book_id,
    borrow_date,
    due_date,
    return_date,
    renew_count,
    status,
    overdue_days,
    overdue_fee,
    fine_amount,
    notes,
    create_time,
    update_time
)
SELECT 
    CONCAT('BR', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(ROW_NUMBER() OVER (ORDER BY old_borrow.id), 4, '0')) AS record_id,
    -- 需要用户ID映射
    (SELECT user_id FROM library1.4.user WHERE username = old_user.username LIMIT 1) AS user_id,
    -- 需要图书ID映射
    (SELECT book_id FROM library1.4.book WHERE isbn = old_book.isbn LIMIT 1) AS book_id,
    borrow_date,
    due_date,
    return_date,
    COALESCE(renew_count, 0) AS renew_count,
    CASE status
        WHEN 'borrowed' THEN 'BORROWED'
        WHEN 'returned' THEN 'RETURNED'
        WHEN 'overdue' THEN 'OVERDUE'
        WHEN 'lost' THEN 'LOST'
        ELSE 'BORROWED'
    END AS status,
    COALESCE(overdue_days, 0) AS overdue_days,
    COALESCE(overdue_fee, 0) AS overdue_fee,
    COALESCE(fine_amount, 0) AS fine_amount,
    notes,
    create_time,
    update_time
FROM old_library.old_borrow
JOIN old_library.old_user ON old_borrow.user_id = old_user.id
JOIN old_library.old_book ON old_borrow.book_id = old_book.id;

-- 2. 验证迁移结果
SELECT COUNT(*) AS total_records FROM library1.4.borrow_record;
SELECT status, COUNT(*) AS count FROM library1.4.borrow_record GROUP BY status;
```

### 3.5 操作日志数据迁移（可选）

```sql
-- 操作日志数据迁移（可选，通常不需要迁移）
-- 如果保留历史日志，可以迁移最近3个月的日志

INSERT INTO library1.4.operation_log (
    log_id,
    user_id,
    operation_type,
    operation_desc,
    ip_address,
    result,
    create_time
)
SELECT 
    CONCAT('LOG', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(ROW_NUMBER() OVER (ORDER BY old_log.id), 4, '0')) AS log_id,
    (SELECT user_id FROM library1.4.user WHERE username = old_user.username LIMIT 1) AS user_id,
    operation_type,
    operation_desc,
    ip_address,
    result,
    create_time
FROM old_library.old_log
WHERE create_time >= DATE_SUB(NOW(), INTERVAL 3 MONTH);
```

## 5. 数据验证

### 5.1 数据完整性验证

```sql
-- 检查数据总数
SELECT 
    'user' AS table_name, COUNT(*) AS count FROM user
UNION ALL
SELECT 'book', COUNT(*) FROM book
UNION ALL
SELECT 'category', COUNT(*) FROM category
UNION ALL
SELECT 'borrow_record', COUNT(*) FROM borrow_record;

-- 检查必填字段
SELECT COUNT(*) AS missing_user_id FROM user WHERE user_id IS NULL;
SELECT COUNT(*) AS missing_isbn FROM book WHERE isbn IS NULL;
SELECT COUNT(*) AS missing_category_id FROM book WHERE category_id IS NULL;
```

### 5.2 数据一致性验证

```sql
-- 检查外键关联
SELECT COUNT(*) AS invalid_user_id 
FROM borrow_record br 
LEFT JOIN user u ON br.user_id = u.user_id 
WHERE u.user_id IS NULL;

SELECT COUNT(*) AS invalid_book_id 
FROM borrow_record br 
LEFT JOIN book b ON br.book_id = b.book_id 
WHERE b.book_id IS NULL;

SELECT COUNT(*) AS invalid_category_id 
FROM book b 
LEFT JOIN category c ON b.category_id = c.category_id 
WHERE c.category_id IS NULL;

-- 检查数据范围
SELECT COUNT(*) AS invalid_price FROM book WHERE price < 0;
SELECT COUNT(*) AS invalid_stock FROM book WHERE stock < 0;
SELECT COUNT(*) AS invalid_available_stock FROM book WHERE available_stock < 0;
```

### 5.3 数据准确性验证

```sql
-- 检查重复数据
SELECT isbn, COUNT(*) AS count 
FROM book 
GROUP BY isbn 
HAVING count > 1;

SELECT username, COUNT(*) AS count 
FROM user 
GROUP BY username 
HAVING count > 1;

-- 检查逻辑一致性
-- 可用库存应该小于等于总库存
SELECT COUNT(*) AS invalid_available_stock 
FROM book 
WHERE available_stock > stock;

-- 借阅记录状态一致性
SELECT COUNT(*) AS invalid_status 
FROM borrow_record 
WHERE status = 'RETURNED' AND return_date IS NULL;
```

## 9. 注意事项

1. **数据备份**
   - 迁移前必须完整备份源数据库
   - 备份文件要保存至少30天

2. **迁移时间**
   - 建议在业务低峰期进行迁移
   - 提前通知用户系统维护时间

3. **数据映射**
   - 确保字段映射关系正确
   - 处理数据类型转换
   - 处理枚举值映射

4. **外键关联**
   - 确保外键关联正确
   - 处理孤立数据

5. **性能影响**
   - 大数据量迁移可能影响性能
   - 考虑分批迁移
   - 监控系统资源使用
